/*
Striver explanation and code from tuf
https://takeuforward.org/data-structure/topological-sort-algorithm-dfs-g-21/

NOTE: Should be acyclic graph - otherwise topological sort is not possible

Problem Statement: Given a Directed Acyclic Graph (DAG) with V vertices and E edges, 
Find any Topological Sorting of that Graph. ORDERING MAINTAINED
Note: In topological sorting, node u appears before node v if there is an edge from node u towards node v(u -> v).
u appears before v in that ordering


Logic - How ordering is maintained
1->2->3->4->5
DFS traversal will end in a terminal node as there will be no adjacent nodes for the terminal node
At 5, vis[4] == 1, and no remaining adjacents
stack.push(5) - Now we add 5 to the stack
stack.push(4) - backtrack and add 4 to the stack
stack.push(3) - backtrack and add 3 to the stack - stack top to bottom = [3,4,5] so u -> direction stays

========================================================================================
Cue -> DFS, mark vis, dfs on adjacents, stack push when no adjacents left
Steps
1. Init visited array, Size V, Init Stack
2. For every node, Run DFS traversal (as there can be non connected components)
3. DFS Function - dfs(node, Stack<Integer> st, List<List<Integer>> adjMatrix, int[] vis)
    - mark vis[node] = 1,
    - Traverse all adjacent nodes, 
        - if not visited, recur dfs again
    - backtrack Step - Traversal breaks on terminal node, no unvisited adjacents present
        - Add the current node to the stack
    vis[node] = 1;
    for(int it: adjMatrix.get(node)) {
        if(vis[it] == 0) {
            dfs(node, stack, adjMatrix, vis);
        }
    }
    st.push(node);
4. Empty the stack to get the answer
========================================================================================

*/

import java.util.*;

class Solution {
    private static void dfs(int node, int vis[], Stack<Integer> st, ArrayList<ArrayList<Integer>> adj) {
        
        //inside the dfs, mark vis array as 1
        //run dfs for every adjacency node
        //once dfs is run mark the node to stack
        vis[node] = 1;
        for (int it : adj.get(node)) {
            if (vis[it] == 0)
                dfs(it, vis, st, adj);
        }
        st.push(node);
    }

    // Function to return list containing vertices in Topological order.
    static int[] topoSort(int V, ArrayList<ArrayList<Integer>> adj) {
        int vis[] = new int[V];
        //maintain a stack
        Stack<Integer> st = new Stack<Integer>();
        //for every visited array element run a dfs
        
        for (int i = 0; i < V; i++) {
            if (vis[i] == 0) {
                dfs(i, vis, st, adj);
            }
        }

        //empty the stack, to get the answer
        int ans[] = new int[V];
        int i = 0;
        while (!st.isEmpty()) {
            ans[i++] = st.peek();
            st.pop();
        }
        return ans;
    }
}

public class tUf {
    public static void main(String[] args) {
        int V = 6;
        ArrayList<ArrayList<Integer>> adj = new ArrayList<>();
        for (int i = 0; i < V; i++) {
            adj.add(new ArrayList<>());
        }
        adj.get(2).add(3);
        adj.get(3).add(1);
        adj.get(4).add(0);
        adj.get(4).add(1);
        adj.get(5).add(0);
        adj.get(5).add(2);

        int[] ans = Solution.topoSort(V, adj);
        for (int node : ans) {
            System.out.print(node + " ");
        }
        System.out.println("");
    }
}
