/*
Cue - to Detect Cycle - On the same path node has to be Visited again, so maintain pathVisited with visited, on backtracking unmark pathVisited
IMP -> omit the pathVisited after traversing all adjacent nodes of the current node,
       do not omit visited, as we dont want traversal again

IMP to Do BFS - we need to use Kahns Algorithm - Topological Sort

Striver explanation and code
https://takeuforward.org/data-structure/detect-cycle-in-a-directed-graph-using-dfs-g-19/


==============================================================================================================
Steps
1. Init Visited arr, PathVisited arr, both size V
2. For every index, Run DFS traversal (as there can be non connected components), check for any true
3. DFS function - dfs(index, List<List<Integer>> adjMatrix, int[] vis, int[] pathVis)
  - mark vis[index] = 1, pathVis[index] = 1
  - Traverse adjacent nodes
    - if not visited, recur dfs on adjacent nodes and check for true
    - if visited, check if pathVis[adj] == 1, (On the same Path, node was visited again. 
      Yayy we have a cycle detected)
      if(visited[adj] == 0) {
        boolean result = dfs(adj, adjMatrix, vis, pathVis);
        if(result) return true;
      } else { //visited case, check pathVisited is 1
        if(pathVisited[adj] == 1) {
          return true;
        }
      }
    - backtrack unmark pathVisited[adj] == 0, return false, // not able to find cycle


==============================================================================================================
*/


import java.util.*;


class Solution {
    private boolean dfsCheck(int node, ArrayList<ArrayList<Integer>> adj, int vis[], int pathVis[]) {
        // set visited node to 1, and path Visited to 1
        vis[node] = 1; 
        pathVis[node] = 1; 
        
        // traverse for adjacent nodes 
        for(int it : adj.get(node)) {
            // when the node is not visited 
            if(vis[it] == 0) {
                if(dfsCheck(it, adj, vis, pathVis) == true) 
                    return true; 
            }
            // if the node has been previously visited
            // but its pathVisited is true, then its a cycle
            else if(pathVis[it] == 1) {
                return true; 
            }
        }

        //set the pathVis back to 0 once the whole dfs is done and we arent able to find the cycle
        pathVis[node] = 0; 
        return false; 
    }

    // Function to detect cycle in a directed graph.
    public boolean isCyclic(int V, ArrayList<ArrayList<Integer>> adj) {
        int vis[] = new int[V];
        int pathVis[] = new int[V];
        
        for(int i = 0; i<V; i++) {
            if(vis[i] == 0) {
                if(dfsCheck(i, adj, vis, pathVis) == true) return true; 
            }
        }
        return false; 
    }
}

public class tUf {
    public static void main(String[] args) {
        int V = 11;
        ArrayList<ArrayList<Integer>> adj = new ArrayList<>();
        for (int i = 0; i < V; i++) {
            adj.add(new ArrayList<>());
        }
        adj.get(1).add(2);
        adj.get(2).add(3);
        adj.get(3).add(4);
        adj.get(3).add(7);
        adj.get(4).add(5);
        adj.get(5).add(6);
        adj.get(7).add(5);
        adj.get(8).add(9);
        adj.get(9).add(10);
        adj.get(10).add(8);

        Solution obj = new Solution();
        boolean ans = obj.isCyclic(V, adj);
        if (ans)
            System.out.println("True");
        else
            System.out.println("False");

    }
}
