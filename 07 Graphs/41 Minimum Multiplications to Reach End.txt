/*
Problem Statement: Given start, end, and an array arr of n numbers. 
At each step, the start is multiplied by any number in the array and then a mod operation with 100000 is done to get the new start. 
Your task is to find the minimum steps in which the end can be achieved starting from the start. 
IMP - Infinite Supply of Numbers in the array
If it is not possible to reach the end, then return -1.
Mod = 100000, if you exceed the range, you need to mod it to bring back the result from 0 to 10^5 - 1 range

Example 1:
Input:
arr[] = {2, 5, 7}
start = 3
end = 30
Output:
2
Explanation: 
Step 1: 3*2 = 6 % 100000 = 6 
Step 2: 6*5 = 30 % 100000 = 30
Therefore, in minimum 2 multiplications we reach the end number which is treated as a destination node of a graph here.

Example 2:
Input:
arr[] = {3, 4, 65}
start = 7
end = 66175
Output:
4
Explanation: 
Step 1: 7*3 = 21 % 100000 = 21 
Step 2: 21*3 = 6 % 100000 = 63 
Step 3: 63*65 = 4095 % 100000 = 4095 
Step 4: 4095*65 = 266175 % 100000 = 66175
Therefore, in minimum 4 multiplications we reach the end number which is treated as a destination node of a graph here.
*/



import java.util.*; 

class Solution {

  static class Pair {
        int node, steps;

        Pair(int node, int steps) {
            this.node = node;
            this.steps = steps;
        }
    }

    public int minimumMultiplications(int[] arr, int start, int end) {
        // Queue to store pairs of (number, steps)
        Queue<Pair> q = new LinkedList<>();
        q.offer(new Pair(start, 0));

        // Distance array to store minimum multiplications needed to reach a number
        int[] dist = new int[100000];
        Arrays.fill(dist, (int)1e9);
        dist[start] = 0;
        int mod = 100000;

        // BFS traversal
        while (!q.isEmpty()) {
            Pair current = q.poll();
            int node = current.node;
            int steps = current.steps;

            // Try multiplying node with each element in arr
            for (int factor : arr) {
                int num = (factor * node) % mod;

                // If this path gives a lesser number of multiplications, update and enqueue
                if (steps + 1 < dist[num]) {
                    dist[num] = steps + 1;

                    // If we reached the target, return the steps
                    if (num == end) return steps + 1;

                    q.offer(new Pair(num, steps + 1));
                }
            }
        }
        // Return -1 if the end number is unreachable
        return -1;
    }

    public static void main(String[] args) {
        int start = 3, end = 30;
        int[] arr = {2, 5, 7};

        Solution obj = new Solution();
        int ans = obj.minimumMultiplications(arr, start, end);

        System.out.println(ans);
    }
}
