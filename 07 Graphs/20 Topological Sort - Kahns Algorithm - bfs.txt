/*
Striver explanation and code
https://takeuforward.org/data-structure/kahns-algorithm-topological-sort-algorithm-bfs-g-22/
Kahns Algorithm -BFS
Problem Statement: Given a Directed Acyclic Graph (DAG) with V vertices and E edges, 
Find any Topological Sorting of that Graph.
Note: In topological sorting, node u will always appear before node v if there is a directed edge from node u towards node v(u -> v).

==================================================================================
Cue -> Fill indegree, add node with 0 indegree to queue, BFS queue, poll node, add to topo arr, reduce indegree of adjacents, check for 0
Steps
1. Calc indegree array for the graph
    0. Init indegree int arr of size V
    1. Traverse all the nodes, 
    2. For Each Node, check for adjacents, increase indegree of the adjacents
2. Init Queue, add all nodes with indegree 0
3. init topo int arr, index = 0
4. BFS Function, 
    - while Queue is Not Empty, 
    - node = queue.peek, queue.remove,
    - add to topo arr, increase index, index++,
    - For each adjacents, 
        - decrease indegree, check if its zero, add to queue
5. return topo arr
==================================================================================

*/



import java.util.*;

class Solution {
    // Function to return list containing vertices in Topological order.
    static int[] topoSort(int V, ArrayList<ArrayList<Integer>> adj) {
        int indegree[] = new int[V];
        for (int i = 0; i < V; i++) {
            for (int it : adj.get(i)) {
                indegree[it]++;
            }
        }

        Queue<Integer> q = new LinkedList<Integer>();
        ;
        for (int i = 0; i < V; i++) {
            if (indegree[i] == 0) {
                q.add(i);
            }
        }

        int topo[] = new int[V];
        int i = 0;
        while (!q.isEmpty()) {
            int node = q.peek();
            q.remove();
            topo[i] = node; //add the node to the topo sort array
            i = i++;
            
            // node is in your topo sort
            // so please remove it from the indegree
            for (int it : adj.get(node)) {
                indegree[it]--;
                if (indegree[it] == 0) {
                    q.add(it); //reduce the indegree of all the adjacent nodes, if someone reaches 0, push it to the queue
                }
            }
        }

        return topo;
    }
}

public class tUf {
    public static void main(String[] args) {
        int V = 6;
        ArrayList<ArrayList<Integer>> adj = new ArrayList<>();
        for (int i = 0; i < V; i++) {
            adj.add(new ArrayList<>());
        }
        adj.get(2).add(3);
        adj.get(3).add(1);
        adj.get(4).add(0);
        adj.get(4).add(1);
        adj.get(5).add(0);
        adj.get(5).add(2);

        int[] ans = Solution.topoSort(V, adj);
        for (int node : ans) {
            System.out.print(node + " ");
        }
        System.out.println("");
    }
}
