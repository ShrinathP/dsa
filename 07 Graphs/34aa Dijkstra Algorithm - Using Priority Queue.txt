import java.util.ArrayList;
import java.util.PriorityQueue;

//can use class Edge as well
class Pair {
    int node;
    int distance;

    public Pair(int node, int distance) {
        this.node = node;
        this.distance = distance;
    }
}

class Solution {

    static int[] dijkstra(int V, ArrayList<ArrayList<Pair>> adj, int S) {

        // Stores shortest distance from source to every node
        int[] dist = new int[V];
        for (int i = 0; i < V; i++) {
            dist[i] = Integer.MAX_VALUE;
        }
        dist[S] = 0;

        // Min Heap based on shortest distance
        PriorityQueue<Pair> pq =
                new PriorityQueue<>((a, b) -> a.distance - b.distance);

        // Source node at distance 0
        pq.add(new Pair(S, 0));

        while (!pq.isEmpty()) {
            Pair current = pq.poll();
            int u = current.node;
            int currDistance = current.distance;

            // If there is already a lesser distance to this node ignore
            if (currDistance > dist[u]) {
                continue;
            }

            // Traverse all neighbours of u
            for (Pair edge : adj.get(u)) {
                int v = edge.node;
                int weight = edge.distance;

                // Update the distance if dist[u] + weight is less than dist[v]
                if (dist[u] + weight < dist[v]) {
                    dist[v] = dist[u] + weight;
                    // Push updated shortest distance
                    pq.add(new Pair(v, dist[v]));
                }
            }
        }

        return dist;
    }
}
