Explanation:
- initialize queue array start, end = -1
- start points to the first element in the queue.
- end points to the last element.
- curSize points current filled Size of the arr
- enqueue() / push: Adds an element at the rear. If the queue is empty, front and end set to 0, else increment end
- dequeue() / pop: Removes an element from the start. After removing, if the queue becomes empty, both front and rear are reset to -1, else increment the start
- Circular Queue: The rear and front wrap around the array using modulo (%), so that when rear or front reaches the end, they return to the beginning.
*/

import java.util.*;

// Class implementing Queue using Arrays
class ArrayQueue {
    // Array to store queue elements
    private int[] arr;
    // Indices for start and end of the queue
    private int start, end;
    // Current size and maximum size of the queue
    private int currSize, maxSize;


    // Constructor
    public ArrayQueue() {
        arr = new int[10];
        start = -1;
        end = -1;
        currSize = 0;
        maxSize = 10;
    }

    // Method to push an element into the queue
    public void push(int x) {
        // Check if the queue is full
        if (currSize == maxSize) {
            System.out.println("Queue is full\nExiting...");
            System.exit(1);
        }

        // If the queue is empty, initialize start and end to 0
        if (end == -1) {
            start = 0;
            end = 0;
        } 
        else {
            // Circular increment of end, increment end circularly
            end = (end + 1) % maxSize;
        }

        // save the pushed element at this endIndex, increment currSize
        arr[end] = x;
        currSize++;
    }

    // Method to pop an element from the queue
    public int pop() {
        // Check if the queue is empty
        if (start == -1) {
            System.out.println("Queue Empty\nExiting...");
            System.exit(1);
        }
        int popped = arr[start];

        // If the queue has only one element, reset start and end to -1
        if (currSize == 1) {
            start = -1;
            end = -1;
        } 
        else {
            // Circular increment of start, move start ahead
            start = (start + 1) % maxSize;
        }

        currSize--;
        return popped;
    }


    // Method to get the front element of the queue
    public int peek() {
        // Check if the queue is empty
        if (start == -1) {
            System.out.println("Queue is Empty");
            System.exit(1);
        }
        return arr[start];
    }

    // Method to determine whether the queue is empty
    public boolean isEmpty() {
        return (currSize == 0);
    }
}

}
